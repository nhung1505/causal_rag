\footnotesize
\textbf{Điểm hỗ trợ của đường} ($h$ bước, $n_k$ = số rule hỗ trợ bước $k$):\\[2pt]
\centerline{$s_{\text{sup}}=\min\!\Big(1,\ \dfrac{1}{h}\displaystyle\sum_{k=1}^{h}\dfrac{\ln\big(1+\max(1,n_k)\big)}{\ln 4}\Big),\qquad
b_{\text{hop}}=\begin{cases}0{,}18 & h=h^{*}=2\\ 0{,}03\cdot h/h^{*} & \text{ngược lại}\end{cases}$}
\dien bước có nhiều rule hỗ trợ thì tin cậy hơn (bão hòa ở 1); đường đủ số bước được thưởng.

\vspace{0.1cm}
\textbf{Điểm bằng chứng của rule:}\quad
$S_{\text{rule}}=0{,}32\,S_{\text{sem}}+0{,}36\,S_{\text{graph}}+0{,}10\,S_{\text{seed}}
+0{,}07\,S_{\text{direct}}+0{,}15\,S_{\text{priority}}$\\[2pt]
\dien trọng số đồ thị/đường cao để giữ rule ở bước thứ hai dù câu hỏi chủ yếu nhắc sự kiện đầu chuỗi.

\vspace{0.1cm}

rule 47: $0{,}32\cdot0{,}555+0{,}36\cdot0{,}705+0{,}10\cdot0{,}603+0+0{,}15\cdot1{,}0=\mathbf{0{,}642}$ (hạng 2)\\
$P_0$: $s_{\text{sup}}=\ln2/\ln4=0{,}5$ (mỗi bước một rule) $\Rightarrow S(P_0)=0{,}705$.}

\vspace{0.08cm}

\begin{frame}
\frametitle{Phụ lục A2 -- Ngữ nghĩa LegalSCM và quy ước cài đặt}
\footnotesize
$n=|V|$ sự kiện, $p=|\mathcal R|$ rule;\quad
$\mathbf B^{\pm}\in\{0,1\}^{p\times n}$ (thân rule: literal dương/âm),\quad
$\mathbf H^{\pm}\in\{0,1\}^{n\times p}$ (đầu rule),\quad
$\mathbf k=(\mathbf B^{+}+\mathbf B^{-})\mathbf 1$.
\[
\mathbf b=\mathbb I[\mathbf B^{+}\mathbf t+\mathbf B^{-}\mathbf f=\mathbf k],\quad
\mathbf a=\mathbf b\odot(\mathbf 1-\mathbf e),\quad
\mathbf d^{\pm}=\mathbb I[\mathbf H^{\pm}\mathbf a\ge1],\quad
\mathbf x^{F}=\operatorname{Fix}(\Phi_{\mathcal R},\mathbf x^{(0)})
\]
\dien hội trong thân rule, tuyển giữa các rule cùng đầu, lặp tới điểm bất động.

\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Theo bài báo}
\footnotesize
\begin{itemize}
  \item Chỉ có hỗ trợ dương $\rightarrow$ TRUE; chỉ có hỗ trợ âm $\rightarrow$ FALSE.
  \item Không có hỗ trợ hoặc hỗ trợ xung đột $\rightarrow$ UNKNOWN.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Theo cài đặt (dùng trong thực nghiệm)}
\footnotesize
\begin{itemize}
  \item Sự kiện nội sinh (là hệ quả của ít nhất một rule) mặc định FALSE khi không cơ chế nào sinh ra.
  \item Sự kiện gốc UNKNOWN cho đến khi có trong ngữ cảnh; xung đột $\rightarrow$ UNKNOWN.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.12cm}
Kích thước thực nghiệm: 1.814 sự kiện, 2.884 rule; 187 sự kiện có hỗ trợ thay thế; 0 rule hội; 0 ngoại lệ.
\end{frame}

\begin{frame}
\frametitle{Phụ lục A3 -- Can thiệp cấu trúc và chẩn đoán xóa nút}
\footnotesize
Sự kiện trung gian của đường chính: $\mathcal M(P^{*})=\{v_1,\dots,v_{h-1}\}$.

\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Chẩn đoán tô-pô}
\footnotesize
$G^{-m}=G[V\setminus\{m\}]$\\[2pt]
$\rho_{\text{top}}(s,Y\mid m)=\mathbb I\big[Y\ \text{đạt được từ}\ s\ \text{trong}\ G^{-m}\big]$\\[2pt]
Giới hạn 3 hop, 30 đường.
\end{block}
\column{0.48\textwidth}
\begin{block}{Can thiệp cấu trúc $do(X_m=0)$}
\footnotesize
$0\equiv\text{FALSE}$;\quad $\mathcal R^{do(X_m=0)}=\{r\in\mathcal R\mid \operatorname{var}(\operatorname{head}(r))\neq m\}$\\[2pt]
$\mathbf x^{CF(m)}=\operatorname{Fix}\big(\Phi_{\mathcal R^{do(X_m=0)}},\mathbf x^{(0)}[X_m\leftarrow0]\big)$
\end{block}
\end{columns}

\vspace{0.12cm}
\[
\nu(m;Y)=\begin{cases}
\text{NECESSARY}, & x_Y^{F}=1,\ x_Y^{CF(m)}=0,\\
\text{NON\_NECESSARY}, & x_Y^{F}=1,\ x_Y^{CF(m)}=1,\\
\text{chưa xác định}, & \text{các trường hợp còn lại.}
\end{cases}
\]

\begin{itemize}
  \item Chỉ đường chính (path 0) được xác minh cấu trúc.
  \item Can thiệp tương đối với mô hình quy tắc đã mã hóa; không có bước abduction biến ngoại sinh.
\end{itemize}
\end{frame}
