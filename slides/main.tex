\begin{frame}{2. Bước 1: Xây dựng legal causal graph}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\centering
\begin{tikzpicture}[node distance=11mm and 14mm]
  \node[draw=black!70,rectangle,font=\scriptsize] (article) {ARTICLE};
  \node[draw=black!70,rectangle,below=of article,font=\scriptsize] (rule) {RULE $r_j$};
  \node[entity,below left=of rule] (condition) {$u_j$};
  \node[entity,below right=of rule] (effect) {$v_j$};
  \node[draw=black!70,rectangle,right=of rule,font=\scriptsize] (subject) {SUBJECT};
  \draw[flow] (article) -- node[left,font=\tiny]{HAS\_RULE} (rule);
  \draw[flow] (rule) -- node[left,font=\tiny]{HAS\_CONDITION} (condition);
  \draw[flow] (rule) -- node[right,font=\tiny]{HAS\_EFFECT} (effect);
  \draw[flow] (rule) -- node[above,font=\tiny]{HAS\_SUBJECT} (subject);
  \draw[flow,line width=1pt] (condition) -- node[below,font=\scriptsize]{CAUSES} (effect);
\end{tikzpicture}

\vspace{2mm}
\begin{equation*}
r_j:\quad u_j\xrightarrow{\;r_j,\,a_j\;}v_j.
\end{equation*}

\column{0.49\textwidth}
\begin{itemize}\small
  \item Biểu diễn bằng \texttt{NetworkX MultiDiGraph}.
  \item Bốn loại nút: ARTICLE, RULE, EVENT và SUBJECT.
  \item Cạnh CAUSES giữ \texttt{rule\_id}, \texttt{article\_id} và metadata pháp lý.
  \item Nhiều rule có thể cùng hỗ trợ một cặp event mà không mất provenance.
  \item Các cách diễn đạt tương đương dùng chung \texttt{event\_id}.
\end{itemize}

\medskip
\textbf{Hình thành path đa bước}
\begin{equation*}
A\xrightarrow{r_1}B,\quad B\xrightarrow{r_2}C
\quad\Longrightarrow\quad
A\rightarrow B\rightarrow C.
\end{equation*}
\end{columns}

\vfill
{\scriptsize Đầu ra: GraphML/GEXF, thống kê đồ thị và danh sách chuỗi hai hop.}
\end{frame}

\begin{frame}{4. Bước 3: Truy xuất causal path đa bước}
\begin{columns}[T,onlytextwidth]
\column{0.43\textwidth}
\textbf{Luồng truy xuất}
\begin{enumerate}\small
  \item Mã hóa query bằng BGE-M3.
  \item FAISS lấy 100 RULE/EVENT gần nhất.
  \item Chọn EVENT top-8 ($s\geq0.20$) và RULE top-8 ($s\geq0.15$).
  \item Chọn hướng forward, backward hoặc both từ query intent.
  \item BFS tối đa 2 hop, không lặp nút; giữ 30 path/seed.
  \item Hợp rule từ seed, path và direct hits; chọn top-12 evidence.
\end{enumerate}

\column{0.55\textwidth}
\textbf{Điểm causal path}
\begin{equation*}
S_{\mathrm{path}}=\operatorname{clip}_{[0,1]}
\left(
0.54s_{\mathrm{seed}}+0.28s_{\mathrm{sup}}
+b_{\mathrm{bridge}}+b_{\mathrm{hop}}
\right).
\end{equation*}
\begin{equation*}
b_{\mathrm{bridge}}=\min(0.18,0.06N_{\mathrm{bridge}}),
\qquad
b_{\mathrm{hop}}=0.18\ \text{nếu đạt 2 hop}.
\end{equation*}

\medskip
\textbf{Điểm rule evidence}
\begin{equation*}
\begin{aligned}
S_{\mathrm{rule}}={}&0.32S_{\mathrm{semantic}}
+0.36S_{\mathrm{graph}}+0.10S_{\mathrm{seed}}\\
&+0.07S_{\mathrm{direct}}+0.15S_{\mathrm{priority}}.
\end{aligned}
\end{equation*}

\medskip
{\small Trọng số graph/path được đặt cao để bảo toàn rule ở hop thứ hai, ngay cả khi query chủ yếu nhắc tới event đầu chuỗi.}
\end{columns}
\end{frame}

\begin{frame}{5. LegalSCM: Ngữ nghĩa ba giá trị và AND/OR}
\begin{columns}[T,onlytextwidth]
\column{0.53\textwidth}
\textbf{Biểu diễn trạng thái}
\begin{equation*}
X_i\in\{1,0,\bot\},\qquad
(t_i,f_i)=
\begin{cases}
(1,0),&X_i=1,\\
(0,1),&X_i=0,\\
(0,0),&X_i=\bot.
\end{cases}
\end{equation*}

\textbf{Kích hoạt rule}
\begin{equation*}
\mathbf b=\mathbb I[\mathbf B^+\mathbf t+\mathbf B^-\mathbf f=\mathbf k],
\qquad
\mathbf a=\mathbf b\odot(\mathbf 1-\mathbf e).
\end{equation*}
\begin{equation*}
\mathbf d^+=\mathbb I[\mathbf H^+\mathbf a\geq1],
\qquad
\mathbf d^-=\mathbb I[\mathbf H^-\mathbf a\geq1].
\end{equation*}

\column{0.44\textwidth}
\textbf{Conjunction trong một rule}
\begin{equation*}
(A\land B)\rightarrow Y.
\end{equation*}
{\small Rule chỉ kích hoạt khi cả hai literal đều thỏa mãn.}

\medskip
\textbf{Alternative disjunction giữa các rule}
\begin{equation*}
A\rightarrow Y,
\qquad
B\rightarrow Y.
\end{equation*}
{\small Một trong hai rule kích hoạt là đủ để hỗ trợ $Y$.}

\medskip
\begin{equation*}
\mathbf x^F=\operatorname{Fix}
(\Phi_{\mathcal R},\mathbf x^{(0)}).
\end{equation*}
\end{columns}

\vspace{2mm}
{\small\textbf{Giới hạn:} engine hỗ trợ nhiều condition/exception, nhưng dữ liệu hiện tại chủ yếu có một antecedent dương; conjunction chưa được kiểm chứng đầy đủ.}

\vfill
{\tiny Nền tảng: Kleene (1952); Fitting (1985); Sato et al. (2020); Sakama et al. (2021).}
\end{frame}

\begin{frame}{6. Bước 4: Xác minh phụ thuộc loại câu hỏi}
\scriptsize
\begin{tabular}{p{0.18\textwidth}p{0.35\textwidth}p{0.40\textwidth}}
\toprule
\textbf{Route} & \textbf{Dạng câu hỏi} & \textbf{Phương pháp xác minh}\\
\midrule
DIRECT
& ``$A$ có trực tiếp gây ra $Y$?''
& Ground hai endpoint ($\geq0.80$), sau đó kiểm tra cạnh $A\to Y$.\\
FACTUAL
& ``$A$ có dẫn đến $Y$?''
& Gán seed TRUE; yêu cầu outcome TRUE, đủ rule coverage và không conflict.\\
INTERVENTION
& ``Nếu bỏ $m$, $Y$ có còn xảy ra?''
& Ground mediator ($\geq0.80$), chạy $do(X_m=0)$ và suy luận lại fixed point.\\
\bottomrule
\end{tabular}

\vspace{4mm}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\textbf{Structural intervention}
\begin{equation*}
\mathcal R^{do(X_m=0)}
=\{r\in\mathcal R:\operatorname{head}(r)\neq X_m\},
\end{equation*}
\begin{equation*}
\mathbf x^{CF(m)}=\operatorname{Fix}
\left(
\Phi_{\mathcal R^{do(X_m=0)}},
\mathbf x^{(0)}[X_m\leftarrow0]
\right).
\end{equation*}
\begin{equation*}
\nu(m;Y)=
\begin{cases}
\mathrm{NECESSARY},&x_Y^F=1,\ x_Y^{CF(m)}=0,\\
\mathrm{NON\_NECESSARY},&x_Y^F=1,\ x_Y^{CF(m)}=1.
\end{cases}
\end{equation*}

\column{0.47\textwidth}
\textbf{Node-deletion diagnostic}
\begin{equation*}
G^{-m}=G[V\setminus\{m\}],
\qquad
\rho_{\mathrm{top}}=
\mathbb I[Y\text{ reachable in }G^{-m}].
\end{equation*}
\begin{itemize}\small
  \item Chỉ tìm alternative path trong giới hạn 3 hop, 30 path.
  \item Không thay thế structural world-state.
  \item Mặc định chỉ Step-3 path 0 được structural verification.
\end{itemize}
\begin{equation*}
\boxed{G^{-m}\neq do(X_m=0)}
\end{equation*}
\end{columns}

\vfill
{\tiny Structural intervention theo Pearl (2009).}
\end{frame}

  \node[process,text width=2.2cm,right=11mm of guard,yshift=-7mm] (fallback) {Không đạt\\Step-3 context};
  \node[process,text width=2.0cm,right=14mm of guard] (llm) {Common LLM\\answer + citation};
  \draw[flow] (verdict) -- (guard);
  \draw[flow] (guard.east) -- (pass.west);
  \draw[flow] (guard.east) -- (fallback.west);
  \draw[flow] (pass.east) -- (llm.west);
  \draw[flow] (fallback.east) -- (llm.west);
\end{tikzpicture}

\vspace{4mm}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\textbf{Điều kiện commit}
\begin{itemize}\small
  \item Primary path hợp lệ và khớp context Step 3.
  \item Verdict khác \textsc{uncertain}; score $\geq0.55$.
  \item Không sử dụng structural fallback bị chặn.
  \item DIRECT: endpoint grounding $\geq0.80$.
  \item FACTUAL: outcome TRUE, đủ hop, không conflict.
  \item INTERVENTION: $do(X_m=0)$ thực thi và world-state xác định.
\end{itemize}

\column{0.47\textwidth}
\textbf{Generation contract}
\begin{itemize}\small
  \item Input gồm question, selected path/evidence, decision và intervention audit.
  \item Không sinh điều luật hoặc citation ngoài context.
  \item Citation được giới hạn trong tập article được cung cấp.
  \item Nếu gate không đạt, common generator quyết định từ nguyên Step-3 evidence.
\end{itemize}
\end{columns}

\vfill
{\scriptsize Trong protocol chính: cùng Qwen3-8B, prompt, JSON schema, parser và seed; extractive fallback bị tắt.}
\end{frame}

\begin{frame}{10. Kiểm định thống kê và diễn giải}
\begin{columns}[T,onlytextwidth]

\textbf{Full LegalSCM so với Causal Path RAG}
\begin{itemize}\small
  \item Số câu đúng: 242 so với 222.
  \item Full-only: 23; Causal-only: 3.
  \item Cả hai đúng: 219; cả hai sai: 5.
  \item $\Delta\mathrm{Acc}=+8.0$ điểm.
  \item Bootstrap CI 95\%: $[4.4,12.0]$.
  \item Exact McNemar: $p=8.8\times10^{-5}$; Holm-adjusted $p=0.0011$.
\end{itemize}

\medskip
\textbf{Audit}
\begin{itemize}\small
  \item Factual / direct / intervention: 230 / 13 / 7.
  \item Verifier committed: 230; Step-3 fallback: 20.
  \item Cả 8 lỗi của Full xảy ra sau no-primary-path fallback.
\end{itemize}

\column{0.48\textwidth}
\textbf{Phạm vi kết luận}
\begin{itemize}\small
  \item Selective verification cải thiện có ý nghĩa structured decision.
  \item Answer F1, ROUGE-L và Citation F1 không khác biệt có ý nghĩa (Holm $p=1.0$).
  \item Cả Full và Causal Path đều đúng 7/7 intervention cases; chưa tách được lợi ích riêng của $do(X_m=0)$.
  \item Top-1 Exact Path 38.56\%, Oracle 75.42\%; path ranking vẫn là nút thắt.
  \item Runtime: 3.238 so với 1.947 giây ($\approx66\%$ overhead).
\end{itemize}

\medskip
\textbf{Hướng tiếp theo:} mở rộng benchmark intervention và multi-antecedent; cải thiện first-hop retrieval, path ranking và grounding.
\end{columns}
\end{frame}
